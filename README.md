Just at the basic levels.

<div style="background-color:#0d1117;padding:16px;border-radius:10px;">

<span style="color:#79c0ff;"># Initialize repository</span>

<span style="color:#7ee787;">git init</span>

<span style="color:#79c0ff;"># Check repository status</span>

<span style="color:#7ee787;">git status</span>

<span style="color:#79c0ff;"># Stage files</span>

<span style="color:#7ee787;">git add .</span>

<span style="color:#79c0ff;"># Commit changes</span>

<span style="color:#7ee787;">git commit -m "Initial commit"</span>

<span style="color:#79c0ff;"># Create and switch to feature branch</span>

<span style="color:#7ee787;">git checkout -b feature-login</span>

<span style="color:#79c0ff;"># Stage and commit feature changes</span>

<span style="color:#7ee787;">git add .</span>

<span style="color:#7ee787;">git commit -m "Add login feature"</span>

<span style="color:#79c0ff;"># Switch back to main</span>

<span style="color:#7ee787;">git checkout main</span>

<span style="color:#79c0ff;"># Merge feature branch into main</span>

<span style="color:#d2a8ff;">git merge feature-login</span>

</div>
